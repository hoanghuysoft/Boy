const ShowListScreen = () => {
  const router = useRouter();
  const { colors, typography, theme } = useThemeContext();
  const common = createCommonStyles(colors, typography);
  const inset = useSafeAreaInsets();

  const ListHeaderComponent = useCallback(
    () => (
      <View style={{ marginTop: inset.top + 66, paddingHorizontal: 16, marginBottom: 8 }}>
        <FlexText style={[common.heading, { fontSize: 28 }]}>
          All shows
        </FlexText>
        <FlexText style={[common.bodySmall, { color: colors.muted, marginTop: 2 }]}>
          Browse everything available on WeLove TV.
        </FlexText>
      </View>
    ),
    [inset.top, common, colors.muted],
  );

  const renderItem = useCallback(
    ({ item }: { item: ShowItem }) => (
      <Pressable
        onPress={() => router.push("/show_detail")}
        style={({ pressed }) => [
          styles.row,
          {
            backgroundColor: colors.background,
            opacity: pressed ? 0.9 : 1,
          },
        ]}
      >
        <Image
          source={{ uri: item.imageUrl }}
          style={styles.thumbnail}
          contentFit="cover"
        />
        <View style={{ flex: 1, gap: 4 }}>
          <FlexText
            numberOfLines={1}
            style={[styles.title, { color: colors.text }]}
          >
            {item.title}
          </FlexText>
          <FlexText
            numberOfLines={2}
            style={[
              styles.description,
              {
                color:
                  // fall back if theme doesn't provide a subtle text color
                  (colors as any).mutedText ?? "rgba(255,255,255,0.7)",
              },
            ]}
          >
            {item.description}
          </FlexText>
        </View>
      </Pressable>
    ),
    [colors, router],
  );

  return (
    <View style={common.screen}>
      {/* Floating back button, same pattern as Bookmarks */}
      <View style={{ position: "absolute", top: inset.top + 16, left: 16, zIndex: 1000 }}>
        <TouchableOpacity
          onPress={() => router.back()}
          style={{
            backgroundColor: colors.containerContent,
            borderRadius: 999,
            width: 40,
            height: 40,
            justifyContent: "center",
            alignItems: "center",
          }}
        >
          <Ionicons name="chevron-back" size={24} color={colors.text} />
        </TouchableOpacity>
      </View>

      <FlatList
        data={DATA}
        keyExtractor={(item) => item.id}
        ListHeaderComponent={ListHeaderComponent}
        contentContainerStyle={{ paddingHorizontal: 16, paddingBottom: inset.bottom + 24, gap: 12 }}
        showsVerticalScrollIndicator={false}
        renderItem={renderItem}
      />
    </View>
  );
};

export default ShowListScreen;

  },
